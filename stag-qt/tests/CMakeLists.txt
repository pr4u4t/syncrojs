function(stag_add_test name)
    qt_add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE stagcore stagnet staglogic Qt6::Test)
    target_compile_definitions(${name} PRIVATE STAG_SQLITE="${CMAKE_SOURCE_DIR}/stag.sqlite" STAG_REFERENCE_DATA="${STAG_REFERENCE_DATA}")
    add_test(NAME ${name} COMMAND ${name})
endfunction()
